    19→    19→
    20→    20→export function AuthPage() {
    21→    21→  const auth = useAuth();
    22→    22→  const [view, setView] = useState<View>(auth.user ? 'family-select' : 'sign-in');
    23→    23→
    24→    24→  // Silently ask for geolocation permission at landing. The browser shows its
    25→    25→  // native prompt (no app-level banner). Result is cached in localStorage so
    26→    26→  // PersonForm can later pre-fill the birthplace field. No visible UI.
    27→    27→  useLandingLocation();
    28→    28→
    29→    29→  // QuickAccess is a one-shot flow: user enters code -> we auto-create a guest
    30→    30→  // account -> we join the family. While this flow is mid-flight (after signUp
    31→    31→  // sets auth.user but before joinFamily sets activeFamily), we MUST keep
    32→    32→  // showing the QuickAccess screen — otherwise AuthPage would re-render and
    33→    33→  // route the now-signed-in user to the "Create a family / Join with a code"
    34→    34→  // intermediary screen, defeating the whole point of Quick Access.
    35→    35→  if (view === 'quick-access' && !auth.activeFamily) {
    36→    36→    return <QuickAccess auth={auth} setView={setView} />;
    37→    37→  }
    38→    38→
    39→    39→  if (!auth.user) {
    40→    40→    return <AuthForms initialView={view} setView={setView} />;
    41→    41→  }
    42→    42→  // If user has families, go straight to family-select (or the active family)
    43→    43→  if (auth.families.length > 0 && auth.activeFamily) {
    44→    44→    // The page.tsx will render FamilyTree since auth.user and auth.activeFamily are set
    45→    45→    return null;
    46→    46→  }
    47→    47→  if (auth.families.length === 0 && view !== 'family-create' && view !== 'family-join') {
    48→    48→    return <FamilyCreateOrJoin setView={setView} />;
    49→    49→  }
    50→    50→  if (view === 'family-create') {
    51→    51→    return <CreateFamily setView={setView} />;
    52→    52→  }
    53→    53→  if (view === 'family-join') {
    54→    54→    return <JoinFamily setView={setView} />;
    55→    55→  }
    56→    56→  return <FamilySelect setView={setView} />;
    57→    57→}

    59→    59→// ---------- Quick Access (family code only, for elders) ----------
    60→    60→function QuickAccess({ auth, setView }: { auth: ReturnType<typeof useAuth>; setView: (v: View) => void }) {
    61→    61→  const [code, setCode] = useState('');
    62→    62→  const [submitting, setSubmitting] = useState(false);
    63→    63→  const [error, setError] = useState<string | null>(null);
    64→    64→
    65→    65→  const handleQuickAccess = async (e: React.FormEvent) => {
    66→    66→    e.preventDefault();
    67→    67→    setError(null);
    68→    68→    const trimmed = code.trim().toUpperCase();
    69→    69→    if (!trimmed) { setError('Enter your family code'); return; }
    70→    70→    setSubmitting(true);
    71→    71→    try {
    72→    72→      const guestId = Math.random().toString(36).slice(2, 10);
    73→    73→      const guestEmail = `guest_${guestId}@familytree.local`;
    74→    74→      const guestPassword = `Guest_${guestId}!`;
    75→    75→      // Sign up (auto-confirmed since mailer_autoconfirm=true)
    76→    76→      const signUpRes = await auth.signUp(guestEmail, guestPassword);
    77→    77→      if (signUpRes.error) {
    78→    78→        // If signup fails (e.g., email exists), try signing in
    79→    79→        const signInRes = await auth.signIn(guestEmail, guestPassword);
    80→    80→        if (signInRes.error) { setError(signInRes.error); setSubmitting(false); return; }
    81→    81→      }
    82→    82→      // Wait for auth state to propagate (user should be set now)
    83→    83→      await new Promise(r => setTimeout(r, 500));
    84→    84→      // Join the family with the code
    85→    85→      const joinRes = await auth.joinFamily(trimmed);
    86→    86→      if (joinRes.error) { setError(joinRes.error); setSubmitting(false); return; }
    87→    87→      // The auth context will now have the family set as active,
    88→    88→      // and page.tsx will render FamilyTree directly — no "Create a family" screen.
    89→    89→    } catch (e: any) { setError(e.message ?? 'Something went wrong'); }
    90→    90→    setSubmitting(false);
    91→    91→  };
    92→    92→
    93→    93→  return (
    94→    94→    <div className="min-h-screen bg-gradient-to-br from-emerald-50 via-white to-teal-50">
    95→    95→      <div className="flex min-h-screen items-center justify-center p-4">
    96→    96→        <div className="w-full max-w-md">
    97→    97→          <div className="mb-6 text-center">
    98→    98→            <div className="mx-auto mb-3 flex h-14 w-14 items-center justify-center rounded-2xl bg-gradient-to-br from-emerald-500 to-teal-600 shadow-lg">
    99→    99→              <TreePine className="h-7 w-7 text-white" />
   100→   100→            </div>
   101→   101→            <h1 className="text-2xl font-bold text-slate-800">Family Tree</h1>
   102→   102→            <p className="text-sm text-slate-500 mt-1">Enter your family code to get started</p>
   103→   103→          </div>
   104→   104→          <div className="rounded-2xl bg-white/80 p-6 shadow-xl backdrop-blur-md ring-1 ring-slate-200">
   105→   105→            <form onSubmit={handleQuickAccess} className="space-y-4" noValidate>
   106→   106→              <div>
   107→   107→                <Label htmlFor="familyCode">Family code</Label>
   108→   108→                <Input id="familyCode" value={code} onChange={(e) => setCode(e.target.value.toUpperCase())} placeholder="AB12CD" className="font-mono text-center text-lg font-bold tracking-[0.3em] uppercase" maxLength={8} autoFocus required />
   109→   109→                <p className="mt-1.5 text-xs text-slate-400">Ask your family member for the 6-character code.</p>
   110→   110→              </div>
   111→   111→              {error && <div className="flex items-start gap-2 rounded-md bg-red-50 px-3 py-2 text-sm text-red-700"><AlertCircle className="mt-0.5 h-4 w-4 shrink-0" /><span>{error}</span></div>}
   112→   112→              <Button type="submit" disabled={submitting} className="w-full bg-gradient-to-r from-emerald-600 to-teal-500 hover:from-emerald-700 hover:to-teal-600">{submitting ? 'Joining…' : 'Enter Family Tree'}</Button>
   113→   113→            </form>
   114→   114→          </div>
   115→   115→          <button onClick={() => { setView('sign-in'); setError(null); }} className="mt-4 w-full text-center text-xs text-slate-500 hover:text-emerald-600">← Sign in with email instead</button>
   116→   116→        </div>
   117→   117→      </div>
   118→   118→    </div>
   119→   119→  );
   120→   120→}

   123→   123→function AuthForms({
   124→   124→  initialView,
   125→   125→  setView,
   126→   126→}: {
   127→   127→  initialView: View;
   128→   128→  setView: (v: View) => void;
   129→   129→}) {
   130→   130→  const auth = useAuth();
   131→   131→  const [isSignUp, setIsSignUp] = useState(initialView === 'sign-up');
   132→   132→  const [email, setEmail] = useState('');
   133→   133→  const [password, setPassword] = useState('');
   134→   134→  const [confirm, setConfirm] = useState('');
   135→   135→  const [submitting, setSubmitting] = useState(false);
   136→   136→  const [error, setError] = useState<string | null>(null);
   137→   137→  const [info, setInfo] = useState<string | null>(null);
   138→   138→  // Magic link (OTP) mode
   139→   139→  const [useMagicLink, setUseMagicLink] = useState(false);
   140→   140→  const [showPassword, setShowPassword] = useState(false);
   141→   141→  const [showConfirm, setShowConfirm] = useState(false);
   142→   142→  const [otpSent, setOtpSent] = useState(false);
   143→   143→  const [otpCode, setOtpCode] = useState('');
   144→   144→
   145→   145→  const submit = async (e: React.FormEvent) => {
   146→   146→    e.preventDefault();
   147→   147→    setError(null);
   148→   148→    setInfo(null);
   149→   149→    if (!email.trim()) {
   150→   150→      setError('Email is required');
   151→   151→      return;
   152→   152→    }
   153→   153→    // Magic link flow
   154→   154→    if (useMagicLink) {
   155→   155→      if (!otpSent) {
   156→   156→        // Step 1: send OTP
   157→   157→        setSubmitting(true);
   158→   158→        const res = await auth.signInWithOtp(email.trim());
   159→   159→        setSubmitting(false);
   160→   160→        if (res.error) {
   161→   161→          setError(res.error);
   162→   162→        } else {
   163→   163→          setOtpSent(true);
   164→   164→          setInfo(`We sent a 6-digit code to ${email.trim()}. Check your inbox and enter it below.`);
   165→   165→        }
   166→   166→        return;
   167→   167→      }
   168→   168→      // Step 2: verify OTP
   169→   169→      if (!otpCode.trim()) {
   170→   170→        setError('Enter the 6-digit code from your email');
   171→   171→        return;
   172→   172→      }
   173→   173→      setSubmitting(true);
   174→   174→      const res = await auth.verifyOtp(email.trim(), otpCode.trim());
   175→   175→      setSubmitting(false);
   176→   176→      if (res.error) {
   177→   177→        setError(res.error);
   178→   178→      } else {
   179→   179→        setInfo('Signed in! Loading your families…');
   180→   180→      }
   181→   181→      return;
   182→   182→    }
   183→   183→    // Password flow
   184→   184→    if (!password) {
   185→   185→      setError('Password is required');
   186→   186→      return;
   187→   187→    }
   188→   188→    if (isSignUp && password !== confirm) {
   189→   189→      setError('Passwords do not match');
   190→   190→      return;
   191→   191→    }
   192→   192→    if (isSignUp && password.length < 6) {
   193→   193→      setError('Password must be at least 6 characters');
   194→   194→      return;
   195→   195→    }
   196→   196→    setSubmitting(true);
   197→   197→    const res = isSignUp
   198→   198→      ? await auth.signUp(email.trim(), password)
   199→   199→      : await auth.signIn(email.trim(), password);
   200→   200→    setSubmitting(false);
   201→   201→    if (res.error) {
   202→   202→      setError(res.error);
   203→   203→    } else {
   204→   204→      setView('family-select');
   205→   205→    }
   206→   206→  };
   207→   207→
   208→   208→  return (
   209→   209→    <div className="min-h-screen bg-gradient-to-br from-emerald-50 via-white to-teal-50">
   210→   210→      <div className="flex min-h-screen items-center justify-center p-4">
   211→   211→        <div className="w-full max-w-md">
   212→   212→          {/* Brand header */}
   213→   213→          <div className="mb-6 text-center">
   214→   214→            <div className="mx-auto mb-3 flex h-14 w-14 items-center justify-center rounded-2xl bg-gradient-to-br from-emerald-500 to-teal-600 shadow-lg">
   215→   215→              <TreePine className="h-7 w-7 text-white" />
   216→   216→            </div>
   217→   217→            <h1 className="text-2xl font-bold text-slate-800">Family Tree</h1>
   218→   218→            <p className="text-sm text-slate-500">
   219→   219→              Build, visualize, and share your family's story
   220→   220→            </p>
   221→   221→            {!isSupabaseConfigured && (
   222→   222→              <div className="mt-3 inline-flex items-center gap-1.5 rounded-full bg-teal-100 px-3 py-1 text-xs font-medium text-teal-700">
   223→   223→                <Sparkles className="h-3 w-3" />
   224→   224→                Demo mode — sign in with any email to explore
   225→   225→              </div>
   226→   226→            )}
   227→   227→          </div>
   228→   228→
   229→   229→          {/* Quick family code access — prominent, above sign-in */}
   230→   230→          {!otpSent && !useMagicLink && (
   231→   231→            <div className="mb-4 mt-4 rounded-2xl border-2 border-emerald-200 bg-emerald-50/50 p-5 text-center">
   232→   232→              <p className="mb-3 text-base font-semibold text-emerald-700">Have a family code?</p>
   233→   233→              <button type="button" onClick={() => { setView('quick-access'); setError(null); }} className="w-full rounded-xl bg-gradient-to-r from-emerald-600 to-teal-500 px-4 py-3 text-lg font-bold text-white shadow-md transition hover:from-emerald-700 hover:to-teal-600 hover:shadow-lg">🔑 Enter Family Code</button>
   234→   234→              <p className="mt-2 text-xs text-slate-400">Simplest way to join — no email needed</p>
   235→   235→            </div>
   236→   236→          )}
   237→   237→
   238→   238→          <div className="rounded-2xl bg-white/80 p-6 shadow-xl backdrop-blur-md ring-1 ring-slate-200">
   239→   239→            {/* Sign-in / Sign-up toggle — hidden in magic link mode */}
   240→   240→            {!useMagicLink && (
   241→   241→              <div className="mb-4 flex rounded-lg bg-slate-100 p-1">
   242→   242→                <button
   243→   243→                  type="button"
   244→   244→                  onClick={() => { setIsSignUp(false); setError(null); }}
   245→   245→                  className={`flex-1 rounded-md px-3 py-1.5 text-sm font-medium transition ${
   246→   246→                    !isSignUp ? 'bg-white shadow text-emerald-700' : 'text-slate-600'
   247→   247→                  }`}
   248→   248→                >
   249→   249→                  Sign in
   250→   250→                </button>
   251→   251→                <button
   252→   252→                  type="button"
   253→   253→                  onClick={() => { setIsSignUp(true); setError(null); }}
   254→   254→                  className={`flex-1 rounded-md px-3 py-1.5 text-sm font-medium transition ${
   255→   255→                    isSignUp ? 'bg-white shadow text-emerald-700' : 'text-slate-600'
   256→   256→                  }`}
   257→   257→                >
   258→   258→                  Create account
   259→   259→                </button>
   260→   260→              </div>
   261→   261→            )}
   262→   262→
   263→   263→            <form onSubmit={submit} className="space-y-3" noValidate>
   264→   264→              <div>
   265→   265→                <Label htmlFor="email">Email</Label>
   266→   266→                <Input
   267→   267→                  id="email"
   268→   268→                  type="email"
   269→   269→                  value={email}

   275→   275→                />
   276→   276→              </div>
   277→   277→
   278→   278→              {/* Magic link OTP code input (shown after sending) */}

   285→   285→                    inputMode="numeric"

   287→   287→                    maxLength={6}

   290→   290→                    placeholder="123456"
   291→   291→                    className="text-center text-lg font-bold tracking-[0.5em]"
   292→   292→                    autoFocus
   293→   293→                  />
   294→   294→                  <button

   297→   297→                    className="mt-1 text-xs text-slate-500 hover:text-slate-700"
